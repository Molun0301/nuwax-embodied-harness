cmake_minimum_required(VERSION 3.16)
project(nuwax_slam_toolbox LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
find_package(Eigen3 3.3 REQUIRED NO_MODULE)
find_package(Boost 1.83.0 EXACT REQUIRED COMPONENTS serialization filesystem thread)
find_package(TBB REQUIRED)
find_package(Ceres 2.2 REQUIRED)
add_library(nuwax_slam_toolbox SHARED
  upstream/karto/src/Karto.cpp upstream/karto/src/Mapper.cpp bridge.cpp solver.cpp)
target_include_directories(nuwax_slam_toolbox PRIVATE upstream/karto/include upstream)
target_link_libraries(nuwax_slam_toolbox PRIVATE Eigen3::Eigen Boost::serialization
  Boost::filesystem Boost::thread TBB::tbb Ceres::ceres)
target_compile_definitions(nuwax_slam_toolbox PRIVATE KARTO_DYNAMIC)
set_target_properties(nuwax_slam_toolbox PROPERTIES VERSION 1.0.0 SOVERSION 1
  INSTALL_RPATH "$ORIGIN" BUILD_RPATH_USE_ORIGIN TRUE)
option(NUWAX_SLAM_SANITIZE "启用原生内存和未定义行为检查" OFF)
if(NUWAX_SLAM_SANITIZE)
  target_compile_options(nuwax_slam_toolbox PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(nuwax_slam_toolbox PRIVATE -fsanitize=address,undefined)
endif()
install(TARGETS nuwax_slam_toolbox LIBRARY DESTINATION lib)
install(FILES bridge.h DESTINATION include/nuwax)
